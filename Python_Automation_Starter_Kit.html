Getting Started with Python

Install Python, set up your environment, and run your first script.

Step 1: Install Python

Download the latest version of Python from the official website. Make sure to check the box that says "Add Python to PATH" during installation.

  1. Go to python.org/downloads
  2. Download Python 3.12+ for your OS (Windows/Mac/Linux)
  3. Run the installer — check "Add Python to PATH"
  4. Click Install Now

Step 2: Verify Installation

Open your terminal (Command Prompt on Windows, Terminal on Mac/Linux) and run:

Terminal 
python --version
Output
Python 3.12.4
If you see a version number, you're all set! If not, restart your computer and try again.

Step 3: Install a Code Editor

We recommend Visual Studio Code (VS Code) — it's free, lightweight, and perfect for Python.

  1. Download from code.visualstudio.com
  2. Install the Python extension from the Extensions tab
  3. Create a new file called hello.py

Your First Script

hello.py 
print("Hello, World! 🚀")
print("Welcome to the Python Automation Starter Kit!")
print("Let's build something amazing together.")

Run it from the terminal:

Terminal 
python hello.py
Output
Hello, World! 🚀 Welcome to the Python Automation Starter Kit! Let's build something amazing together.
You just ran your first Python script! Every chapter from here will build on this foundation.
Chapter 02

Auto-Organize Your Files

Automatically sort messy downloads folders by file type.

Tired of a cluttered Downloads folder? This script automatically moves files into folders based on their type — images go to Images/, PDFs go to Documents/, and so on.

How It Works

  • Scans every file in the target folder
  • Identifies the file extension (.jpg, .pdf, .mp4, etc.)
  • Creates category folders if they don't exist
  • Moves each file to the correct folder automatically

The Script

file_organizer.py 
import os
import shutil

# Change this to your target folder
TARGET_DIR = os.path.expanduser("~/Downloads")

# Define file categories and their extensions
CATEGORIES = {
    "Images":    [".jpg", ".jpeg", ".png", ".gif", ".svg", ".webp"],
    "Documents": [".pdf", ".doc", ".docx", ".txt", ".xlsx", ".pptx"],
    "Videos":    [".mp4", ".mkv", ".avi", ".mov"],
    "Audio":     [".mp3", ".wav", ".flac", ".aac"],
    "Archives":  [".zip", ".rar", ".7z", ".tar", ".gz"],
    "Code":      [".py", ".js", ".html", ".css", ".json"],
}

def organize_files():
    moved_count = 0

    for filename in os.listdir(TARGET_DIR):
        filepath = os.path.join(TARGET_DIR, filename)

        # Skip directories
        if os.path.isdir(filepath):
            continue

        # Get file extension
        _, ext = os.path.splitext(filename)
        ext = ext.lower()

        # Find the right category
        destination_folder = "Others"
        for category, extensions in CATEGORIES.items():
            if ext in extensions:
                destination_folder = category
                break

        # Create folder if needed & move file
        dest_path = os.path.join(TARGET_DIR, destination_folder)
        os.makedirs(dest_path, exist_ok=True)
        shutil.move(filepath, os.path.join(dest_path, filename))
        moved_count += 1
        print(f"  ✅ {filename} → {destination_folder}/")

    print(f"\n🎉 Done! Organized {moved_count} files.")

if __name__ == "__main__":
    print("📂 Organizing your files...\n")
    organize_files()
Sample Output
📂 Organizing your files... ✅ photo_2024.jpg → Images/ ✅ assignment.pdf → Documents/ ✅ lecture.mp4 → Videos/ ✅ song.mp3 → Audio/ ✅ project.zip → Archives/ 🎉 Done! Organized 5 files.
You can customize the CATEGORIES dictionary to add your own file types and folder names. Add .apk to a new "Apps" category, for example!
Chapter 03

Bulk Rename Files Like a Pro

Rename hundreds of files in seconds with a pattern.

Need to rename 200 photos from IMG_3847.jpg to vacation_001.jpg? This script does it in one click.

The Script

bulk_rename.py 
import os

# Configuration
FOLDER = os.path.expanduser("~/Pictures/Rename")
PREFIX = "photo"        # New name prefix
START_NUM = 1           # Start counting from
PADDING = 3             # Number padding (001, 002, 003...)

def bulk_rename():
    files = sorted(os.listdir(FOLDER))
    count = 0

    for i, filename in enumerate(files, start=START_NUM):
        filepath = os.path.join(FOLDER, filename)

        if os.path.isdir(filepath):
            continue

        _, ext = os.path.splitext(filename)
        new_name = f"{PREFIX}_{str(i).zfill(PADDING)}{ext}"
        new_path = os.path.join(FOLDER, new_name)

        os.rename(filepath, new_path)
        print(f"  📝 {filename} → {new_name}")
        count += 1

    print(f"\n🎉 Renamed {count} files!")

if __name__ == "__main__":
    print("✏️ Bulk Rename Tool\n")
    bulk_rename()
Sample Output
✏️ Bulk Rename Tool 📝 IMG_3847.jpg → photo_001.jpg 📝 IMG_3848.jpg → photo_002.jpg 📝 IMG_3849.jpg → photo_003.jpg 🎉 Renamed 3 files!
Always test on a small folder first. Renaming is hard to undo — keep backups!

Web Scraping — Extract Data from Websites

Pull headlines, prices, or any data from websites automatically.

Install Required Library

Terminal 
pip install requests beautifulsoup4

The Script — Scrape News Headlines

scraper.py 
import requests
from bs4 import BeautifulSoup

URL = "https://news.ycombinator.com/"

def scrape_headlines():
    print("🌐 Fetching headlines...\n")

    response = requests.get(URL)
    soup = BeautifulSoup(response.text, "html.parser")

    # Find all headline links
    headlines = soup.select(".titleline > a")

    for i, headline in enumerate(headlines[:10], 1):
        title = headline.text
        link = headline.get("href", "#")
        print(f"  {i}. {title}")
        print(f"     🔗 {link}\n")

    print(f"✅ Scraped {min(10, len(headlines))} headlines!")

if __name__ == "__main__":
    scrape_headlines()
Sample Output
🌐 Fetching headlines... 1. Show HN: I built an open-source AI tool 🔗 https://example.com/article1 2. Python 3.13 new features explained 🔗 https://example.com/article2 ✅ Scraped 10 headlines!
You can modify the soup.select() line to target any HTML element on any website. Right-click → Inspect Element to find the CSS selector!

import pywhatkit as kit

# Send a message to a phone number at a specific time
# Format: +country_code + number

Email Automation with Python

Send professional emails programmatically — perfect for newsletters and notifications.

The Script

email_sender.py 
import smtplib
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart

# Your Gmail credentials
SENDER_EMAIL = "your.email@gmail.com"
APP_PASSWORD = "your-app-password"  # NOT your Gmail password

# Recipient
TO_EMAIL = "recipient@example.com"

def send_email():
    # Create message
    msg = MIMEMultipart()
    msg["From"] = SENDER_EMAIL
    msg["To"] = TO_EMAIL
    msg["Subject"] = "Hello from Python! 🐍"

    body = """
    Hi there!

    This email was sent automatically using Python.
    Pretty cool, right?

    Best regards,
    Your Python Script 🤖
    """
    msg.attach(MIMEText(body, "plain"))

    # Send email
    print("📧 Sending email...")
    with smtplib.SMTP("smtp.gmail.com", 587) as server:
        server.starttls()
        server.login(SENDER_EMAIL, APP_PASSWORD)
        server.sendmail(SENDER_EMAIL, TO_EMAIL, msg.as_string())

    print("✅ Email sent successfully!")

if __name__ == "__main__":
    send_email()
For Gmail, you need to generate an App Password. Go to Google Account → Security → 2-Step Verification → App Passwords → Generate.

Merge Multiple PDFs

pdf_merger.py 
from PyPDF2 import PdfMerger

def merge_pdfs(file_list, output_name="merged.pdf"):
    merger = PdfMerger()

    for pdf_file in file_list:
        print(f"  📄 Adding: {pdf_file}")
        merger.append(pdf_file)

    merger.write(output_name)
    merger.close()
    print(f"\n✅ Merged into: {output_name}")

# List your PDF files here
files_to_merge = [
    "chapter1.pdf",
    "chapter2.pdf",
    "chapter3.pdf",
]

if __name__ == "__main__":
    print("📑 PDF Merger Tool\n")
    merge_pdfs(files_to_merge)

Extract Specific Pages

pdf_splitter.py 
from PyPDF2 import PdfReader, PdfWriter

def extract_pages(input_pdf, pages, output_name="extracted.pdf"):
    reader = PdfReader(input_pdf)
    writer = PdfWriter()

    for page_num in pages:
        writer.add_page(reader.pages[page_num - 1])
        print(f"  📄 Extracted page {page_num}")

    with open(output_name, "wb") as f:
        writer.write(f)

    print(f"\n✅ Saved to: {output_name}")

# Extract pages 1, 3, and 5 from a PDF
if __name__ == "__main__":
    print("✂️ PDF Splitter Tool\n")
    extract_pages("document.pdf", pages=[1, 3, 5])

Password Generator Script

Generate strong, random passwords for all your accounts.

The Script

password_gen.py 
import random
import string

def generate_password(length=16, use_special=True):
    chars = string.ascii_letters + string.digits
    if use_special:
        chars += string.punctuation

    # Ensure at least one of each type
    password = [
        random.choice(string.ascii_uppercase),
        random.choice(string.ascii_lowercase),
        random.choice(string.digits),
    ]
    if use_special:
        password.append(random.choice(string.punctuation))

    # Fill remaining length
    remaining = length - len(password)
    password.extend(random.choices(chars, k=remaining))

    # Shuffle to randomize position
    random.shuffle(password)
    return "".join(password)

if __name__ == "__main__":
    print("🔐 Password Generator\n")
    print("  Generated passwords:\n")

    for i in range(5):
        pwd = generate_password(length=16)
        print(f"  {i+1}. {pwd}")

    print("\n  ✅ Use a password manager to store them safely!")

System Info & Health Monitor

Check CPU, memory, disk usage, and system health in real-time.

Install Required Library

Terminal 
pip install psutil

The Script

system_monitor.py 
import psutil
import platform
from datetime import datetime

def system_report():
    print("=" * 50)
    print("       💻 SYSTEM HEALTH REPORT")
    print("=" * 50)

    # OS Info
    print(f"\n🖥️  OS: {platform.system()} {platform.release()}")
    print(f"📛 Machine: {platform.machine()}")
    print(f"🕐 Time: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")

    # CPU
    cpu_percent = psutil.cpu_percent(interval=1)
    cpu_count = psutil.cpu_count()
    print(f"\n⚡ CPU Usage: {cpu_percent}%")
    print(f"   Cores: {cpu_count}")

    # Memory
    mem = psutil.virtual_memory()
    print(f"\n🧠 RAM: {mem.used / (1024**3):.1f} GB / {mem.total / (1024**3):.1f} GB")
    print(f"   Usage: {mem.percent}%")

    # Disk
    disk = psutil.disk_usage("/")
    print(f"\n💾 Disk: {disk.used / (1024**3):.1f} GB / {disk.total / (1024**3):.1f} GB")
    print(f"   Usage: {disk.percent}%")

    # Battery (if laptop)
    battery = psutil.sensors_battery()
    if battery:
        status = "⚡ Charging" if battery.power_plugged else "🔋 On Battery"
        print(f"\n🔋 Battery: {battery.percent}% ({status})")

    print("\n" + "=" * 50)
    print("  ✅ System is healthy!" if cpu_percent < 80 else "  ⚠️ CPU usage is high!")
    print("=" * 50)

if __name__ == "__main__":
    system_report()
